export function getHeroExpToNextLevel(level: number): number {
  const safeLvl = Math.max(1, Number(level) || 1);
  return Math.floor(70 * Math.pow(1.15, safeLvl - 1));
}

export function isIdleRpgUnlocked(player: PlayerState): boolean {
  return (
    player?.settings?.miniGamesUnlocked ||
    player?.rank === "S" ||
    player?.rank === "SS" ||
    (Number(player?.level) || 1) >= 40
  );
}

export function computeHeroStatsFromPlayer(
  player: PlayerState,
  saveState: IdleRpgSaveState
): IdleRpgHeroStats {
  const pStats = {
    strength: Number(player?.stats?.strength) || 10,
    agility: Number(player?.stats?.agility) || 10,
    vitality: Number(player?.stats?.vitality) || 10,
    intelligence: Number(player?.stats?.intelligence) || 10,
    perception: Number(player?.stats?.perception) || 10,
  };

  const heroLvl = Math.max(1, Number(saveState?.heroLevel) || 1);
  const up = {
    attackLevel: Math.max(0, Number(saveState?.upgrades?.attackLevel) || 0),
    hpLevel: Math.max(0, Number(saveState?.upgrades?.hpLevel) || 0),
    critLevel: Math.max(0, Number(saveState?.upgrades?.critLevel) || 0),
    speedLevel: Math.max(0, Number(saveState?.upgrades?.speedLevel) || 0),
  };

  // Calculate percentage boosts from Avatar profile
  const percentBoosts: IdleRpgProfilePercentBoosts = {
    strAttackPercent: Math.floor(pStats.strength * 1.5),
    agiCritPercent: Number((pStats.agility * 0.35).toFixed(1)),
    agiSpeedPercent: Math.floor(pStats.agility * 0.4),
    vitHpPercent: Math.floor(pStats.vitality * 2.0),
    vitRegenPercent: Math.floor(pStats.vitality * 1.5),
    intSkillDmgPercent: Math.floor(pStats.intelligence * 1.8),
    intMpRegenPercent: Math.floor(pStats.intelligence * 1.4),
    perGoldPercent: Math.floor(pStats.perception * 1.6),
    perDropPercent: Math.floor(pStats.perception * 0.8),
    auraPowerPercent: Math.floor(((Number(player?.combatPower) || 100) / 10) + ((Number(player?.level) || 1) * 2)),
  };

  // Calculate Equipment flat + % bonuses
  let eqAtk = 0;
  let eqDef = 0;
  let eqHp = 0;
  let eqCrit = 0;

  if (saveState?.equipped) {
    Object.values(saveState.equipped).forEach((item) => {
      if (item) {
        const itemLvl = Math.max(0, Number(item.level) || 0);
        const multiplier = 1 + itemLvl * 0.18;
        eqAtk += Math.floor((Number(item.attack) || 0) * multiplier);
        eqDef += Math.floor((Number(item.defense) || 0) * multiplier);
        eqHp += Math.floor((Number(item.hp) || 0) * multiplier);
        eqCrit += (Number(item.critRate) || 0) + itemLvl * 0.5;
      }
    });
  }

  // Base raw values before % modifiers
  const rawAttack = 40 + heroLvl * 8 + up.attackLevel * 14 + eqAtk;
  const rawHp = 500 + heroLvl * 75 + up.hpLevel * 130 + eqHp;
  const rawDef = 14 + heroLvl * 2.5 + Math.floor(pStats.vitality * 0.5) + eqDef;
  const rawMp = 100 + heroLvl * 12 + Math.floor(pStats.intelligence * 2.5);

  // Apply Percentage Multipliers from Profile
  const strMult = 1 + percentBoosts.strAttackPercent / 100;
  const vitMult = 1 + percentBoosts.vitHpPercent / 100;
  const auraMult = 1 + percentBoosts.auraPowerPercent / 100;

  const finalAttack = Math.floor(rawAttack * strMult * auraMult);
  const finalHp = Math.floor(rawHp * vitMult);
  const finalDef = Math.floor(rawDef);
  const finalMp = Math.floor(rawMp);

  const critRate = Math.min(80, Math.floor(8 + percentBoosts.agiCritPercent + up.critLevel * 1.5 + eqCrit));
  const critMultiplier = Number((1.5 + pStats.agility * 0.007).toFixed(2));
  const attackSpeedMs = Math.max(500, 1500 - Math.floor(pStats.agility * 9) - up.speedLevel * 30);

  const hpRegenPerSec = Math.max(6, Math.floor((10 + heroLvl * 2.5) * (1 + percentBoosts.vitRegenPercent / 100)));
  const mpRegenPerSec = Math.max(5, Math.floor((8 + heroLvl * 1.5) * (1 + percentBoosts.intMpRegenPercent / 100)));
  const skillDmgMultiplier = Number((1.0 + percentBoosts.intSkillDmgPercent / 100).toFixed(2));
  const goldMultiplier = Number((1.0 + percentBoosts.perGoldPercent / 100).toFixed(2));
  const dropRateMultiplier = Number((1.0 + percentBoosts.perDropPercent / 100).toFixed(2));

  return {
    level: heroLvl,
    maxHp: finalHp,
    currentHp: finalHp,
    hpRegenPerSec,
    attack: finalAttack,
    defense: finalDef,
    critRate,
    critMultiplier,
    attackSpeedMs,
    maxMp: finalMp,
    currentMp: finalMp,
    mpRegenPerSec,
    skillDmgMultiplier,
    goldMultiplier,
    dropRateMultiplier,
    auraPower: Number(auraMult.toFixed(2)),
    percentBoosts,
  };
}

export function generateStageEnemy(stage: number): IdleRpgEnemy {
  const stg = Math.max(1, Number(stage) || 1);
  const isBoss = stg % 5 === 0;
  const chapter = Math.floor((stg - 1) / 5) + 1;
  const stageInChapter = ((stg - 1) % 5) + 1;

  const mobNames = [
    "Krwawy Pająk",
    "Upiór Otchłani",
    "Lodowy Okruchowiec",
    "Mroczny Wilkołak",
    "Zguba Cienia",
    "Watażka Demonów",
  ];

  const bossNames = [
    "Kamish Władca Smoków",
    "Władca Bramy Niebios",
    "Baran Król Demonów",
    "Tytan Zniszczenia",
  ];

  const name = isBoss
    ? `[BOSS ${chapter}-5] ${bossNames[(chapter - 1) % bossNames.length]}`
    : `${mobNames[(stg - 1) % mobNames.length]} (Brama ${chapter}-${stageInChapter})`;

  const scalingFactor = Math.pow(1.06, Math.min(100, stg - 1));
  const bossMult = isBoss ? 3.2 : 1.0;

  const maxHp = Math.floor((220 + stg * 45) * scalingFactor * bossMult);
  const attack = Math.floor((15 + stg * 3.5) * Math.pow(1.04, Math.min(100, stg - 1)) * (isBoss ? 1.6 : 1.0));
  const defense = Math.floor((5 + stg * 1.2) * Math.pow(1.03, Math.min(100, stg - 1)));

  const goldReward = Math.floor((25 + stg * 8) * (isBoss ? 5.0 : 1.0));
  const expReward = Math.floor((18 + stg * 6) * (isBoss ? 4.5 : 1.0));

  return {
    id: `enemy_stage_${stg}_${Date.now()}`,
    name,
    isBoss,
    maxHp,
    currentHp: maxHp,
    attack,
    defense,
    goldReward,
    expReward,
    spriteKey: isBoss ? "boss" : "mob",
  };
}

export function generateEquipmentDrop(stage: number, isBoss: boolean): IdleRpgEquipment | null {
  const dropChance = isBoss ? 1.0 : 0.32;
  if (Math.random() > dropChance) return null;

  const slots: IdleRpgEquipmentSlot[] = ["weapon", "armor", "gloves", "boots", "accessory"];
  const slot = slots[Math.floor(Math.random() * slots.length)];

  // Rarity roll
  const roll = Math.random() * 100;
  let rarity: IdleRpgRarity = "common";
  if (isBoss) {
    if (roll < 12) rarity = "mythic";
    else if (roll < 38) rarity = "legendary";
    else if (roll < 78) rarity = "epic";
    else rarity = "rare";
  } else {
    if (roll < 3) rarity = "legendary";
    else if (roll < 14) rarity = "epic";
    else if (roll < 42) rarity = "rare";
    else rarity = "common";
  }

  const rarityMultipliers: Record<IdleRpgRarity, number> = {
    common: 1.0,
    rare: 1.6,
    epic: 2.6,
    legendary: 4.4,
    mythic: 7.2,
  };
  const mult = rarityMultipliers[rarity] * Math.pow(1.04, Math.min(100, Math.max(1, Number(stage) || 1)));

  const weaponIcons = [
    "/assets/idlerpg/sprites/eq_dagger_shadow.png",
    "/assets/idlerpg/sprites/eq_sword_flame.png",
    "/assets/idlerpg/sprites/eq_sword_frost.png",
    "/assets/idlerpg/sprites/eq_scythe_death.png",
    "/assets/idlerpg/sprites/eq_katana_blood.png",
    "/assets/idlerpg/sprites/eq_greatsword_monarch.png",
    "/assets/idlerpg/sprites/eq_daggers_kamish.png",
  ];

  let name = "";
  let icon = "";
  let attack = 0;
  let defense = 0;
  let hp = 0;
  let critRate = 0;

  if (slot === "weapon") {
    name = rarity === "mythic" ? "Sztylety Gniewu Kamisha" : rarity === "legendary" ? "Ostrze Władcy Cieni" : "Mroczny Sztylet Łowcy";
    icon = weaponIcons[Math.floor(Math.random() * weaponIcons.length)];
    attack = Math.floor(30 * mult);
    critRate = Math.floor(3 + mult);
  } else if (slot === "armor") {
    name = rarity === "mythic" ? "Zbroja Monarchy Cieni" : "Pancerz Bramy Cienia";
    icon = "/assets/idlerpg/sprites/eq_chest_armor.png";
    defense = Math.floor(15 * mult);
    hp = Math.floor(100 * mult);
  } else if (slot === "gloves") {
    name = "Rękawice Szybkości Cienia";
    icon = "/assets/idlerpg/sprites/eq_gloves_speed.png";
    attack = Math.floor(12 * mult);
    critRate = Math.floor(4 * mult);
  } else if (slot === "boots") {
    name = "Buty Zwinności Monarchy";
    icon = "/assets/idlerpg/sprites/eq_boots_agility.png";
    defense = Math.floor(10 * mult);
    hp = Math.floor(65 * mult);
  } else {
    name = rarity === "mythic" ? "Amulet Władcy Otchłani" : "Pierścień Mrocznej Energii";
    icon = Math.random() > 0.5 ? "/assets/idlerpg/sprites/eq_ring_shadow.png" : "/assets/idlerpg/sprites/eq_amulet_monarch.png";
    attack = Math.floor(18 * mult);
    defense = Math.floor(10 * mult);
    hp = Math.floor(75 * mult);
  }

  return {
    id: `eq_${Date.now()}_${Math.floor(Math.random() * 1000)}`,
    name,
    slot,
    rarity,
    level: 0,
    attack,
    defense,
    hp,
    critRate,
    icon,
  };
}

export function computeEquipmentUpgradeCost(item: IdleRpgEquipment): { gold: number; crystals: number } {
  const itemLvl = Math.max(0, Number(item?.level) || 0);
  return {
    gold: Math.floor(120 * Math.pow(1.35, itemLvl)),
    crystals: itemLvl >= 3 ? Math.floor(1 + itemLvl * 0.8) : 0,
  };
}

export function computeItemSellPrice(item: IdleRpgEquipment): number {
  const rarityPrices: Record<IdleRpgRarity, number> = {
    common: 60,
    rare: 180,
    epic: 500,
    legendary: 1500,
    mythic: 4000,
  };
  const base = rarityPrices[item.rarity] || 60;
  const upgradeRefund = Math.floor((item.level || 0) * 80);
  return base + upgradeRefund;
}

export function getNextAutoBattleSkill(
  skills: IdleRpgSkill[],
  currentMp: number,
  maxMp: number,
  isBoss: boolean,
  battleSpeed: number
): IdleRpgSkill | null {
  const now = Date.now();
  const readySkills = skills.filter((s) => {
    const cd = s.cooldownMs / (battleSpeed || 1);
    return now - s.lastUsedAtMs >= cd && currentMp >= s.manaCost;
  });

  if (readySkills.length === 0) return null;

  if (isBoss) {
    // If Monarch Awakening (Ultimate) is ready, unleash it immediately!
    const ultimate = readySkills.find((s) => s.isUltimate);
    if (ultimate) return ultimate;

    // Highest multiplier burst skill against Boss
    return readySkills.sort((a, b) => b.multiplier - a.multiplier)[0];
  } else {
    // Regular mob: conserve mana if below 30% Max MP for upcoming Boss
    const mpPercent = currentMp / Math.max(1, maxMp);
    if (mpPercent < 0.3) {
      // Only cast cheap skill
      const cheap = readySkills.filter((s) => s.manaCost <= 35 && !s.isUltimate);
      return cheap.length > 0 ? cheap[0] : null;
    }

    // Do not waste ultimate on normal mobs
    const nonUltimates = readySkills.filter((s) => !s.isUltimate);
    if (nonUltimates.length > 0) {
      return nonUltimates.sort((a, b) => a.manaCost - b.manaCost)[0];
    }
  }

  return null;
}

export function calculateOfflineAfkRewards(
  saveState: IdleRpgSaveState,
  heroStats: IdleRpgHeroStats
): AfkRewardSummary {
  const now = Date.now();
  const lastExit = Number(saveState?.lastActiveExitAt) || Number(saveState?.lastAfkClaimAt) || now;
  const elapsedMs = Math.max(0, now - lastExit);

  const chestLvl = Math.max(1, Number(saveState?.afkChestLevel) || 1);
  const maxHours = getAfkChestMaxHours(chestLvl);
  const maxStorageMs = maxHours * 3600 * 1000;
  const cappedElapsedMs = Math.min(elapsedMs, maxStorageMs);
  const elapsedMinutes = cappedElapsedMs / (60 * 1000);

  const chestMultiplier = 1 + (chestLvl - 1) * 0.25;
  const stage = Math.max(1, Number(saveState?.stage) || 1);
  const baseGoldPerMin = (18 + stage * 5.0) * (Number(heroStats?.goldMultiplier) || 1) * chestMultiplier;
  const baseExpPerMin = (12 + stage * 3.5) * chestMultiplier;

  const goldEarned = Math.floor(elapsedMinutes * baseGoldPerMin);
  const expEarned = Math.floor(elapsedMinutes * baseExpPerMin);
  const manaCrystalsEarned = Math.floor((elapsedMinutes / 25) * (1 + (chestLvl - 1) * 0.15));

  return {
    elapsedSeconds: Math.floor(cappedElapsedMs / 1000),
    maxStorageSeconds: Math.floor(maxStorageMs / 1000),
    isFull: elapsedMs >= maxStorageMs,
    goldEarned,
    expEarned,
    manaCrystalsEarned,
    goldPerMinute: Math.floor(baseGoldPerMin),
    expPerMinute: Math.floor(baseExpPerMin),
  };
}

export function computeUpgradeCost(
  type: keyof IdleRpgSaveState["upgrades"],
  currentLevel: number
): number {
  const baseCosts = {
    attackLevel: 50,
    hpLevel: 40,
    critLevel: 100,
    speedLevel: 150,
  };
  const lvl = Math.max(0, Number(currentLevel) || 0);
  return Math.floor(baseCosts[type] * Math.pow(1.18, lvl));
}

export function computeShadowUpgradeCost(
  shadowId: keyof IdleRpgSaveState["shadowUpgrades"],
  currentLevel: number
): { gold: number; crystals: number } {
  const base = {
    igris_knight: { gold: 120, crystals: 2 },
    shadow_wolf: { gold: 250, crystals: 5 },
    shadow_mage: { gold: 500, crystals: 10 },
    kaisel_dragon: { gold: 1200, crystals: 25 },
    shadow_assassin: { gold: 2500, crystals: 50 },
  };

  const config = base[shadowId] || { gold: 200, crystals: 5 };
  const lvl = Math.max(0, Number(currentLevel) || 0);
  return {
    gold: Math.floor(config.gold * Math.pow(1.22, lvl)),
    crystals: Math.floor(config.crystals * Math.pow(1.15, lvl)),
  };
}

export function loadIdleRpgSave(): IdleRpgSaveState {
  const def = createDefaultIdleRpgSave();
  try {
    const raw = localStorage.getItem(IDLE_RPG_STORAGE_KEY);
    if (!raw) return def;
    const parsed = JSON.parse(raw);
    return {
      version: 3,
      stage: Math.max(1, Number(parsed.stage) || 1),
      maxStage: Math.max(1, Number(parsed.maxStage) || 1),
      gold: Math.max(0, Number(parsed.gold) || 0),
      manaCrystals: Math.max(0, Number(parsed.manaCrystals) || 10),
      heroExp: Math.max(0, Number(parsed.heroExp) || 0),
      heroLevel: Math.max(1, Number(parsed.heroLevel) || 1),
      skillPoints: Math.max(0, Number(parsed.skillPoints) || 0),
      afkChestLevel: Math.max(1, Number(parsed.afkChestLevel) || 1),
      lastAfkClaimAt: Number(parsed.lastAfkClaimAt) || Date.now(),
      lastActiveExitAt: Number(parsed.lastActiveExitAt) || Date.now(),
      respawnAvailableAtMs: Number(parsed.respawnAvailableAtMs) || 0,
      autoBattle: parsed.autoBattle !== false,
      battleSpeed: parsed.battleSpeed === 2 || parsed.battleSpeed === 5 ? parsed.battleSpeed : 1,
      unlockedShadows: Array.isArray(parsed.unlockedShadows) ? parsed.unlockedShadows : ["igris_knight"],
      equipped: parsed.equipped && typeof parsed.equipped === "object" ? parsed.equipped : {},
      inventory: Array.isArray(parsed.inventory) ? parsed.inventory : [],
      skills: parsed.skills && typeof parsed.skills === "object" ? parsed.skills : def.skills,
      upgrades: {
        attackLevel: Math.max(0, Number(parsed.upgrades?.attackLevel) || 0),
        hpLevel: Math.max(0, Number(parsed.upgrades?.hpLevel) || 0),
        critLevel: Math.max(0, Number(parsed.upgrades?.critLevel) || 0),
        speedLevel: Math.max(0, Number(parsed.upgrades?.speedLevel) || 0),
      },
      shadowUpgrades: {
        igris_knight: Math.max(0, Number(parsed.shadowUpgrades?.igris_knight) || 0),
        shadow_wolf: Math.max(0, Number(parsed.shadowUpgrades?.shadow_wolf) || 0),
        shadow_mage: Math.max(0, Number(parsed.shadowUpgrades?.shadow_mage) || 0),
        kaisel_dragon: Math.max(0, Number(parsed.shadowUpgrades?.kaisel_dragon) || 0),
        shadow_assassin: Math.max(0, Number(parsed.shadowUpgrades?.shadow_assassin) || 0),
      },
      stats: {
        totalMonstersSlain: Math.max(0, Number(parsed.stats?.totalMonstersSlain) || 0),
        totalBossesDefeated: Math.max(0, Number(parsed.stats?.totalBossesDefeated) || 0),
        totalGoldEarned: Math.max(0, Number(parsed.stats?.totalGoldEarned) || 0),
        totalDamageDealt: Math.max(0, Number(parsed.stats?.totalDamageDealt) || 0),
      },
    };
  } catch {
    return def;
  }
}

export function saveIdleRpgSave(state: IdleRpgSaveState): void {
  try {
    localStorage.setItem(IDLE_RPG_STORAGE_KEY, JSON.stringify(state));
  } catch {}
}
