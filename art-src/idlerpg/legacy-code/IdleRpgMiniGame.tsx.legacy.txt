export function IdleRpgMiniGame({ player, onClose, onRewardPlayer }: IdleRpgMiniGameProps) {
  const [saveState, setSaveState] = useState<IdleRpgSaveState>(() => loadIdleRpgSave());
  const [activeTab, setActiveTab] = useState<"battle" | "hero" | "equipment" | "skills" | "shadows" | "afk">("battle");
  const [enemy, setEnemy] = useState<IdleRpgEnemy>(() => generateStageEnemy(saveState.stage));
  const [heroCurrentHp, setHeroCurrentHp] = useState(1000);
  const [heroCurrentMp, setHeroCurrentMp] = useState(200);
  const [floatingTexts, setFloatingTexts] = useState<FloatingText[]>([]);
  const [heroAttacking, setHeroAttacking] = useState(false);
  const [enemyHitFlash, setEnemyHitFlash] = useState(false);
  const [heroHitFlash, setHeroHitFlash] = useState(false);
  const [slashSparkActive, setSlashSparkActive] = useState<string | null>(null);
  const [afkSummary, setAfkSummary] = useState<AfkRewardSummary | null>(null);
  const [claimToast, setClaimToast] = useState<string | null>(null);
  const [comboCount, setComboCount] = useState(0);
  const [awakeningActive, setAwakeningActive] = useState(false);
  const [stageClearBanner, setStageClearBanner] = useState<string | null>(null);
  const [levelUpBanner, setLevelUpBanner] = useState<number | null>(null);
  const [selectedEqItem, setSelectedEqItem] = useState<IdleRpgEquipment | null>(null);
  const [nowTime, setNowTime] = useState(Date.now());

  // Skills with level calculations
  const [rawSkills, setRawSkills] = useState<IdleRpgSkill[]>(() => getDefaultSkills());

  const heroStats: IdleRpgHeroStats = useMemo(
    () => computeHeroStatsFromPlayer(player, saveState),
    [player, saveState]
  );

  const shadowArmy: IdleRpgShadowSoldier[] = useMemo(
    () => computeShadowArmy(player, saveState),
    [player, saveState]
  );

  const skills: IdleRpgSkill[] = useMemo(() => {
    return rawSkills.map((base) => {
      const savedLevel = saveState.skills?.[base.id]?.level || 1;
      return getSkillEffectiveData(base, savedLevel);
    });
  }, [rawSkills, saveState.skills]);

  // Sync timer
  useEffect(() => {
    const timer = setInterval(() => setNowTime(Date.now()), 500);
    return () => clearInterval(timer);
  }, []);

  // Save changes & record last active timestamp on unload
  useEffect(() => {
    saveIdleRpgSave(saveState);
  }, [saveState]);

  useEffect(() => {
    return () => {
      // Save last active exit timestamp when component unmounts
      setSaveState((prev) => {
        const next = { ...prev, lastActiveExitAt: Date.now() };
        saveIdleRpgSave(next);
        return next;
      });
    };
  }, []);

  // Sync HP/MP on stats upgrade
  useEffect(() => {
    setHeroCurrentHp((prev) => Math.min(prev || heroStats.maxHp, heroStats.maxHp) || heroStats.maxHp);
    setHeroCurrentMp((prev) => Math.min(prev || heroStats.maxMp, heroStats.maxMp) || heroStats.maxMp);
  }, [heroStats.maxHp, heroStats.maxMp]);

  // Calculate offline AFK rewards summary
  useEffect(() => {
    setAfkSummary(calculateOfflineAfkRewards(saveState, heroStats));
  }, [saveState, heroStats]);

  // Floating damage number helper
  const addDamageText = (damage: number, isCrit: boolean, isHeroDamage = false) => {
    const newText: FloatingText = {
      id: Date.now() + Math.random(),
      text: isCrit ? `CRITICAL -${damage.toLocaleString()}` : `-${damage.toLocaleString()}`,
      isCrit,
      x: isHeroDamage ? 18 + Math.random() * 12 : 64 + Math.random() * 12,
      y: 28 + Math.random() * 18,
      color: isHeroDamage ? "#f87171" : isCrit ? "#fbbf24" : "#38bdf8",
    };
    setFloatingTexts((prev) => [...prev.slice(-8), newText]);
  };

  const isDead = saveState.respawnAvailableAtMs > 0 && nowTime < saveState.respawnAvailableAtMs;
  const canFreeRespawn = saveState.respawnAvailableAtMs > 0 && nowTime >= saveState.respawnAvailableAtMs;
  const respawnSecondsLeft = Math.max(0, Math.ceil((saveState.respawnAvailableAtMs - nowTime) / 1000));

  // Level Up Check
  const checkLevelUp = (currentExp: number, currentLevel: number) => {
    let exp = currentExp;
    let lvl = currentLevel;
    let gainedPoints = 0;
    let expRequired = getHeroExpToNextLevel(lvl);

    while (exp >= expRequired) {
      exp -= expRequired;
      lvl += 1;
      gainedPoints += 1;
      expRequired = getHeroExpToNextLevel(lvl);
    }

    if (lvl > currentLevel) {
      setLevelUpBanner(lvl);
      setTimeout(() => setLevelUpBanner(null), 3000);
      setClaimToast(`🌟 AWANS JEDNOSTKI! Poziom Łowcy Lv.${lvl} (+${gainedPoints} SP)`);
      setTimeout(() => setClaimToast(null), 3500);
      return { nextExp: exp, nextLvl: lvl, pointsGained: gainedPoints };
    }
    return { nextExp: exp, nextLvl: lvl, pointsGained: 0 };
  };

  // Combat Loop
  useEffect(() => {
    if (isDead) return;

    let lastHeroAttack = Date.now();
    let lastShadowAttack = Date.now();
    let lastEnemyAttack = Date.now();
    let lastRegen = Date.now();

    const interval = setInterval(() => {
      const now = Date.now();
      const speed = saveState.battleSpeed || 1;

      // Natural HP & MP Regeneration
      if (now - lastRegen >= 1000 / speed) {
        setHeroCurrentHp((prev) => Math.min(heroStats.maxHp, prev + heroStats.hpRegenPerSec));
        setHeroCurrentMp((prev) => Math.min(heroStats.maxMp, prev + heroStats.mpRegenPerSec));
        lastRegen = now;
      }

      // Hero Attack with forward lunge and clean hit spark directly on the monster
      if (now - lastHeroAttack >= heroStats.attackSpeedMs / speed) {
        lastHeroAttack = now;
        setHeroAttacking(true);
        setSlashSparkActive(SPRITES.slashes[Math.floor(Math.random() * SPRITES.slashes.length)]);
        setTimeout(() => {
          setHeroAttacking(false);
          setSlashSparkActive(null);
        }, 160);

        const isCrit = Math.random() * 100 <= heroStats.critRate;
        const multiplier = isCrit ? heroStats.critMultiplier : 1;
        const awakenMultiplier = awakeningActive ? 1.6 : 1;
        const damage = Math.max(
          1,
          Math.floor(
            (heroStats.attack * multiplier * awakenMultiplier * (0.9 + Math.random() * 0.2)) -
              enemy.defense
          )
        );

        addDamageText(damage, isCrit, false);
        setEnemyHitFlash(true);
        setTimeout(() => setEnemyHitFlash(false), 120);
        setComboCount((c) => c + 1);

        setEnemy((curr) => {
          const nextHp = curr.currentHp - damage;
          if (nextHp <= 0) {
            handleEnemyDefeated(curr);
            return generateStageEnemy(saveState.stage);
          }
          return { ...curr, currentHp: nextHp };
        });
      }

      // Shadow Army Attacks
      if (now - lastShadowAttack >= 1200 / speed) {
        lastShadowAttack = now;
        const activeShadows = shadowArmy.filter((s) => s.unlocked);
        if (activeShadows.length > 0) {
          const totalShadowDps = activeShadows.reduce((acc, s) => acc + s.attack, 0);
          const shadowDmg = Math.max(
            1,
            Math.floor(totalShadowDps * (awakeningActive ? 2.0 : 1.0) * (0.85 + Math.random() * 0.3))
          );
          addDamageText(shadowDmg, false, false);
          setEnemyHitFlash(true);
          setTimeout(() => setEnemyHitFlash(false), 100);

          setEnemy((curr) => {
            const nextHp = curr.currentHp - shadowDmg;
            if (nextHp <= 0) {
              handleEnemyDefeated(curr);
              return generateStageEnemy(saveState.stage);
            }
            return { ...curr, currentHp: nextHp };
          });
        }
      }

      // Enemy Attack
      if (now - lastEnemyAttack >= 2000 / speed) {
        lastEnemyAttack = now;
        const enemyDmg = Math.max(1, Math.floor(enemy.attack * (0.9 + Math.random() * 0.2) - heroStats.defense));
        addDamageText(enemyDmg, false, true);
        setHeroHitFlash(true);
        setTimeout(() => setHeroHitFlash(false), 120);

        setHeroCurrentHp((hp) => {
          const next = hp - enemyDmg;
          if (next <= 0) {
            handleHeroDeath();
            return 0;
          }
          return next;
        });
      }

      // Intelligent Auto Battle: Mana Management & Boss Prioritization
      if (saveState.autoBattle) {
        const nextSkill = getNextAutoBattleSkill(
          skills,
          heroCurrentMp,
          heroStats.maxMp,
          enemy.isBoss,
          saveState.battleSpeed || 1
        );
        if (nextSkill) {
          triggerSkill(nextSkill.id);
        }
      }
    }, 100);

    return () => clearInterval(interval);
  }, [heroStats, shadowArmy, enemy, saveState.battleSpeed, saveState.autoBattle, awakeningActive, heroCurrentMp, skills, isDead]);

  const handleHeroDeath = () => {
    setSaveState((prev) => ({
      ...prev,
      respawnAvailableAtMs: Date.now() + RESPAWN_DELAY_MS,
    }));
    setClaimToast("⚠️ ŁOWCA POLEGŁ W BRAMIE! Wybierz sposób odrodzenia.");
  };

  const handleInstantRespawn = () => {
    const hpCost = Math.max(10, Math.floor((player.maxHp || 100) * 0.15));
    if ((player.hp || 100) <= hpCost) {
      setClaimToast("❌ Zbyt mało HP w głównym profilu awatara na odrodzenie krwią!");
      setTimeout(() => setClaimToast(null), 3000);
      return;
    }

    onRewardPlayer?.({ hpCost: -hpCost });
    setSaveState((prev) => ({
      ...prev,
      respawnAvailableAtMs: 0,
    }));
    setHeroCurrentHp(heroStats.maxHp);
    setHeroCurrentMp(heroStats.maxMp);
    setEnemy(generateStageEnemy(saveState.stage));
    setClaimToast("⚡ Odrodzono natychmiastowo poprzez Esencję Cienia!");
    setTimeout(() => setClaimToast(null), 3000);
  };

  const handleFreeRespawn = () => {
    setSaveState((prev) => ({
      ...prev,
      respawnAvailableAtMs: 0,
    }));
    setHeroCurrentHp(heroStats.maxHp);
    setHeroCurrentMp(heroStats.maxMp);
    setEnemy(generateStageEnemy(saveState.stage));
    setClaimToast("✨ Łowca odrodził się w pełni sił!");
    setTimeout(() => setClaimToast(null), 3000);
  };

  const handleEnemyDefeated = (defeatedEnemy: IdleRpgEnemy) => {
    const goldEarned = Math.floor(defeatedEnemy.goldReward * heroStats.goldMultiplier);
    const expEarned = defeatedEnemy.expReward;
    const isBoss = defeatedEnemy.isBoss;

    // Stage transition animation
    setStageClearBanner(isBoss ? `🏆 BOSS POKONANY!` : `⚔️ BRAMA UKOŃCZONA!`);
    setTimeout(() => setStageClearBanner(null), 1200);

    // Roll equipment drop
    const eqDrop = generateEquipmentDrop(saveState.stage, isBoss);

    setSaveState((prev) => {
      const nextStage = prev.stage + 1;
      const nextMax = Math.max(prev.maxStage, nextStage);
      const totalExp = prev.heroExp + expEarned;
      const lvlCheck = checkLevelUp(totalExp, prev.heroLevel);

      const nextInv = eqDrop ? [...prev.inventory.slice(-40), eqDrop] : prev.inventory;

      return {
        ...prev,
        stage: nextStage,
        maxStage: nextMax,
        gold: prev.gold + goldEarned,
        heroExp: lvlCheck.nextExp,
        heroLevel: lvlCheck.nextLvl,
        skillPoints: prev.skillPoints + lvlCheck.pointsGained,
        manaCrystals: isBoss ? prev.manaCrystals + 5 : prev.manaCrystals,
        inventory: nextInv,
        stats: {
          ...prev.stats,
          totalMonstersSlain: prev.stats.totalMonstersSlain + 1,
          totalBossesDefeated: isBoss ? prev.stats.totalBossesDefeated + 1 : prev.stats.totalBossesDefeated,
          totalGoldEarned: prev.stats.totalGoldEarned + goldEarned,
          totalDamageDealt: prev.stats.totalDamageDealt + defeatedEnemy.maxHp,
        },
      };
    });

    if (eqDrop) {
      setClaimToast(`🎁 ZDOBYTO PRZEDMIOT: [${eqDrop.rarity.toUpperCase()}] ${eqDrop.name}!`);
      setTimeout(() => setClaimToast(null), 3500);
    }

    if (isBoss) {
      onRewardPlayer?.({ gold: 150, exp: 80 });
    }
  };

  const triggerSkill = (skillId: IdleRpgSkill["id"]) => {
    const skill = skills.find((s) => s.id === skillId);
    if (!skill || heroCurrentMp < skill.manaCost) return;
    const now = Date.now();
    if (now - skill.lastUsedAtMs < skill.cooldownMs / (saveState.battleSpeed || 1)) return;

    setHeroCurrentMp((mp) => Math.max(0, mp - skill.manaCost));
    setRawSkills((list) =>
      list.map((s) => (s.id === skillId ? { ...s, lastUsedAtMs: now } : s))
    );

    if (skill.isUltimate) {
      setAwakeningActive(true);
      setTimeout(() => setAwakeningActive(false), 8000);
    }

    setSlashSparkActive(SPRITES.slashes[0]);
    setTimeout(() => setSlashSparkActive(null), 250);

    const damage = Math.floor(
      heroStats.attack * skill.multiplier * heroStats.skillDmgMultiplier * (0.95 + Math.random() * 0.2)
    );
    addDamageText(damage, true, false);
    setEnemyHitFlash(true);
    setTimeout(() => setEnemyHitFlash(false), 180);

    setEnemy((curr) => {
      const nextHp = curr.currentHp - damage;
      if (nextHp <= 0) {
        handleEnemyDefeated(curr);
        return generateStageEnemy(saveState.stage);
      }
      return { ...curr, currentHp: nextHp };
    });
  };

  // Stage selector / Rollback
  const handleSelectStage = (newStage: number) => {
    const target = Math.max(1, Math.min(saveState.maxStage, newStage));
    setSaveState((prev) => ({
      ...prev,
      stage: target,
      respawnAvailableAtMs: 0,
    }));
    setEnemy(generateStageEnemy(target));
    setHeroCurrentHp(heroStats.maxHp);
    setHeroCurrentMp(heroStats.maxMp);
  };

  // Equipment operations
  const handleEquipItem = (item: IdleRpgEquipment) => {
    const currentlyEquipped = saveState.equipped[item.slot];
    const newInventory = saveState.inventory.filter((i) => i.id !== item.id);
    if (currentlyEquipped) {
      newInventory.push(currentlyEquipped);
    }

    setSaveState((prev) => ({
      ...prev,
      equipped: {
        ...prev.equipped,
        [item.slot]: item,
      },
      inventory: newInventory,
    }));
    setSelectedEqItem(null);
  };

  const handleUnequipItem = (slot: IdleRpgEquipmentSlot) => {
    const item = saveState.equipped[slot];
    if (!item) return;

    setSaveState((prev) => ({
      ...prev,
      equipped: {
        ...prev.equipped,
        [slot]: undefined,
      },
      inventory: [...prev.inventory, item],
    }));
    setSelectedEqItem(null);
  };

  const handleUpgradeEquipment = (item: IdleRpgEquipment) => {
    const cost = computeEquipmentUpgradeCost(item);
    if (saveState.gold < cost.gold || saveState.manaCrystals < cost.crystals) return;

    const upgraded: IdleRpgEquipment = {
      ...item,
      level: item.level + 1,
    };

    if (saveState.equipped[item.slot]?.id === item.id) {
      setSaveState((prev) => ({
        ...prev,
        gold: prev.gold - cost.gold,
        manaCrystals: prev.manaCrystals - cost.crystals,
        equipped: {
          ...prev.equipped,
          [item.slot]: upgraded,
        },
      }));
    } else {
      setSaveState((prev) => ({
        ...prev,
        gold: prev.gold - cost.gold,
        manaCrystals: prev.manaCrystals - cost.crystals,
        inventory: prev.inventory.map((i) => (i.id === item.id ? upgraded : i)),
      }));
    }
    setSelectedEqItem(upgraded);
  };

  // Sell individual equipment item
  const handleSellItem = (item: IdleRpgEquipment) => {
    const price = computeItemSellPrice(item);
    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold + price,
      inventory: prev.inventory.filter((i) => i.id !== item.id),
    }));
    setSelectedEqItem(null);
    setClaimToast(`💰 Sprzedano ${item.name} za +${price.toLocaleString()} Złota!`);
    setTimeout(() => setClaimToast(null), 3000);
  };

  // Bulk sell common & rare items
  const handleBulkSellJunk = () => {
    const junk = saveState.inventory.filter((i) => i.rarity === "common" || i.rarity === "rare");
    if (junk.length === 0) {
      setClaimToast("Brak pospolitych lub rzadkich przedmiotów do sprzedaży!");
      setTimeout(() => setClaimToast(null), 3000);
      return;
    }

    let totalGain = 0;
    junk.forEach((item) => {
      totalGain += computeItemSellPrice(item);
    });

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold + totalGain,
      inventory: prev.inventory.filter((i) => i.rarity !== "common" && i.rarity !== "rare"),
    }));
    setSelectedEqItem(null);
    setClaimToast(`💰 Sprzedano ${junk.length} przedmiotów za +${totalGain.toLocaleString()} Złota!`);
    setTimeout(() => setClaimToast(null), 3500);
  };

  // Unlock Shadow with Diamonds
  const handleUnlockShadowWithDiamonds = (shadow: IdleRpgShadowSoldier) => {
    if (saveState.manaCrystals < shadow.diamondUnlockCost) {
      setClaimToast(`❌ Potrzebujesz ${shadow.diamondUnlockCost} 💎 do odblokowania ${shadow.name}!`);
      setTimeout(() => setClaimToast(null), 3000);
      return;
    }

    setSaveState((prev) => ({
      ...prev,
      manaCrystals: prev.manaCrystals - shadow.diamondUnlockCost,
      unlockedShadows: [...new Set([...(prev.unlockedShadows || []), shadow.id])],
    }));

    setClaimToast(`👑 ODBLOKOWANO CIENIA: ${shadow.name} dołączył do armii!`);
    setTimeout(() => setClaimToast(null), 3500);
  };

  // Upgrade AFK Chest
  const handleUpgradeAfkChest = () => {
    const currentLevel = saveState.afkChestLevel || 1;
    if (currentLevel >= 10) return;
    const cost = computeAfkChestUpgradeCost(currentLevel);
    if (saveState.gold < cost.gold || saveState.manaCrystals < cost.crystals) {
      setClaimToast(`❌ Potrzebujesz ${cost.gold.toLocaleString()}🪙 i ${cost.crystals}💎 na ulepszenie skrzyni!`);
      setTimeout(() => setClaimToast(null), 3000);
      return;
    }

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold - cost.gold,
      manaCrystals: prev.manaCrystals - cost.crystals,
      afkChestLevel: currentLevel + 1,
    }));

    setClaimToast(`🎁 ULEPSZONO SKRZYNIĘ AFK do Poziomu ${currentLevel + 1}!`);
    setTimeout(() => setClaimToast(null), 3500);
  };

  const handleUpgradeSkill = (skillId: string) => {
    const currentLevel = saveState.skills[skillId]?.level || 1;
    const cost = computeSkillUpgradeCost(currentLevel);
    if (saveState.gold < cost.gold || saveState.skillPoints < cost.skillPoints) return;

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold - cost.gold,
      skillPoints: prev.skillPoints - cost.skillPoints,
      skills: {
        ...prev.skills,
        [skillId]: { level: currentLevel + 1 },
      },
    }));
  };

  const handleClaimAfk = () => {
    if (!afkSummary || afkSummary.goldEarned <= 0) {
      setClaimToast("Brak nagród offline do odebrania w tym momencie!");
      setTimeout(() => setClaimToast(null), 3000);
      return;
    }

    const { goldEarned, expEarned, manaCrystalsEarned } = afkSummary;
    const totalExp = saveState.heroExp + expEarned;
    const lvlCheck = checkLevelUp(totalExp, saveState.heroLevel);

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold + goldEarned,
      heroExp: lvlCheck.nextExp,
      heroLevel: lvlCheck.nextLvl,
      skillPoints: prev.skillPoints + lvlCheck.pointsGained,
      manaCrystals: prev.manaCrystals + manaCrystalsEarned,
      lastAfkClaimAt: Date.now(),
      lastActiveExitAt: Date.now(),
    }));

    onRewardPlayer?.({ gold: Math.floor(goldEarned * 0.25), exp: Math.floor(expEarned * 0.25) });
    setClaimToast(`Odebrano: +${goldEarned.toLocaleString()} Złota, +${expEarned} EXP, +${manaCrystalsEarned} Kryształów!`);
    setTimeout(() => setClaimToast(null), 3500);
  };

  const handleUpgrade = (type: keyof IdleRpgSaveState["upgrades"]) => {
    const cost = computeUpgradeCost(type, saveState.upgrades[type]);
    if (saveState.gold < cost) return;

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold - cost,
      upgrades: {
        ...prev.upgrades,
        [type]: prev.upgrades[type] + 1,
      },
    }));
  };

  const handleShadowUpgrade = (shadowId: keyof IdleRpgSaveState["shadowUpgrades"]) => {
    const cost = computeShadowUpgradeCost(shadowId, saveState.shadowUpgrades[shadowId]);
    if (saveState.gold < cost.gold || saveState.manaCrystals < cost.crystals) return;

    setSaveState((prev) => ({
      ...prev,
      gold: prev.gold - cost.gold,
      manaCrystals: prev.manaCrystals - cost.crystals,
      shadowUpgrades: {
        ...prev.shadowUpgrades,
        [shadowId]: prev.shadowUpgrades[shadowId] + 1,
      },
    }));
  };

  const chapter = Math.floor((saveState.stage - 1) / 5) + 1;
  const subStage = ((saveState.stage - 1) % 5) + 1;
  const stageWallpaper = CHAPTER_WALLPAPERS[(chapter - 1) % CHAPTER_WALLPAPERS.length];

    if (enemy.isBoss) {
      return SPRITES.bosses[(chapter - 1) % SPRITES.bosses.length];

    return SPRITES.enemies[(saveState.stage + subStage) % SPRITES.enemies.length];
  }, [enemy.isBoss, chapter, saveState.stage, subStage]);

  // Hunter Sprite Selection
  const currentHunterSrc = useMemo(() => {
    if (awakeningActive) return SPRITES.hunter.awakened;
    if (heroAttacking) return SPRITES.hunter.attack;
    return SPRITES.hunter.dual;
  }, [awakeningActive, heroAttacking]);

  const expRequired = getHeroExpToNextLevel(saveState.heroLevel);
  const expPercent = Math.min(100, Math.floor((saveState.heroExp / Math.max(1, expRequired)) * 100));

  const afkChestSprite = getAfkChestSprite(saveState.afkChestLevel || 1);
  const afkMaxHours = getAfkChestMaxHours(saveState.afkChestLevel || 1);

  return (
    <div className="fixed inset-0 z-[150] flex flex-col bg-slate-950 text-slate-100 select-none overflow-hidden font-sans">
      {/* Top Clean Header Bar */}
      <div className="flex items-center justify-between border-b border-cyan-500/30 bg-slate-900/95 px-3 py-1.5 shadow-md">
        <div className="flex items-center gap-2">
          <div className="relative h-8 w-8 rounded-xl border border-cyan-400/40 bg-cyan-950/80 p-0.5 shadow-[0_0_12px_rgba(6,182,212,0.4)] overflow-hidden">
            <img src={SPRITES.hunter.portrait} alt="Hunter" className="h-full w-full object-contain" />
          </div>
          <div>
            <div className="flex items-center gap-1.5">
              <span className="text-xs font-black uppercase tracking-wider text-cyan-300 drop-shadow-[0_0_6px_rgba(6,182,212,0.8)]">
                SHADOW MONARCH IDLE RPG
              </span>
              <span className="rounded bg-gradient-to-r from-amber-500/30 to-purple-500/30 px-1.5 py-0.2 text-[8px] font-black uppercase tracking-widest text-amber-300 border border-amber-400/40">
                RANGA S
              </span>

            <div className="flex items-center gap-1.5 mt-0.5">
              <button
                type="button"
                onClick={() => handleSelectStage(saveState.stage - 1)}
                disabled={saveState.stage <= 1}
                className="flex h-4 w-4 items-center justify-center rounded bg-slate-800 text-slate-300 hover:bg-slate-700 disabled:opacity-30"
              >
                <ChevronLeft className="h-3 w-3" />
              </button>

              <span className="text-[9px] font-bold text-slate-300 font-mono">
                Brama {chapter}-{subStage} · Etap {saveState.stage} (Rekord: {saveState.maxStage})
              </span>

              <button
                type="button"
                onClick={() => handleSelectStage(saveState.stage + 1)}
                disabled={saveState.stage >= saveState.maxStage}
                className="flex h-4 w-4 items-center justify-center rounded bg-slate-800 text-slate-300 hover:bg-slate-700 disabled:opacity-30"
              >
                <ChevronRight className="h-3 w-3" />
              </button>
            </div>
          </div>
        </div>

        {/* Currencies & Controls */}
        <div className="flex items-center gap-1.5">
          {/* Gold */}
          <div className="flex items-center gap-1 rounded-xl bg-slate-950/90 border border-amber-500/30 px-2 py-0.5">
            <span className="text-[11px]">🪙</span>
            <span className="font-mono text-[11px] font-black text-amber-300">{saveState.gold.toLocaleString()}</span>
          </div>
          {/* Mana Crystals */}
          <div className="flex items-center gap-1 rounded-xl bg-slate-950/90 border border-purple-500/30 px-2 py-0.5">
            <span className="text-[11px]">💎</span>
            <span className="font-mono text-[11px] font-black text-purple-300">{saveState.manaCrystals}</span>
          </div>

          {/* Speed Toggle */}
          <button
            type="button"
            onClick={() =>
              setSaveState((prev) => ({
                ...prev,
                battleSpeed: prev.battleSpeed === 1 ? 2 : prev.battleSpeed === 2 ? 5 : 1,
              }))

            className="flex h-7 items-center rounded-xl border border-cyan-400/40 bg-cyan-950/80 px-2 text-[9px] font-black text-cyan-200"

            x{saveState.battleSpeed}

              saveState.autoBattle
                ? "border-emerald-400/60 bg-emerald-950/80 text-emerald-300 shadow-[0_0_8px_rgba(16,185,129,0.4)]"

            {saveState.autoBattle ? "AUTO" : "MANUAL"}

          {/* Close */}
          <button
            type="button"
            onClick={onClose}
            className="flex h-7 w-7 items-center justify-center rounded-xl border border-slate-700 bg-slate-800/80 text-slate-300 hover:bg-slate-700"
          >
            <X className="h-4 w-4" />
          </button>
        </div>
      </div>

      {/* Main 2D Battle Stage Arena */}
      <div className="relative flex-1 min-h-[300px] overflow-hidden bg-slate-950">
        {/* Dynamic Atmospheric Wallpaper */}
        <div
          className="absolute inset-0 bg-cover bg-center opacity-70 filter brightness-90 transition-all duration-700"
          style={{ backgroundImage: `url('${stageWallpaper}')` }}
        />
        <div className="absolute inset-0 bg-gradient-to-t from-slate-950 via-slate-950/20 to-slate-950/60" />

        {/* Awakening Glow FX */}
        {awakeningActive && (
          <div className="absolute inset-0 border-4 border-purple-500/70 pointer-events-none animate-pulse shadow-[inset_0_0_60px_rgba(168,85,247,0.6)]" />
        )}

        {/* Top-Right Solo Leveling RPG HUD Stats Panel (Matching User Uploaded Mockup) */}
        <div className="absolute right-3 top-3 z-20 w-48 rounded-xl border-2 border-purple-600/70 bg-slate-950/90 p-2 shadow-[0_0_20px_rgba(147,51,234,0.5)] backdrop-blur-md space-y-1.5">
          {/* Level Header with Purple Progress Bar */}
          <div className="flex items-center justify-between">
            <span className="text-xs font-black tracking-wider text-purple-300 font-mono">
              Lv. {saveState.heroLevel}
            </span>
            <div className="h-2 w-24 overflow-hidden rounded-full bg-slate-900 border border-purple-900">
              <div
                className="h-full bg-gradient-to-r from-purple-600 to-indigo-400 transition-all duration-200"

          <div className="space-y-0.5">
            <div className="flex items-center justify-between text-[8px] font-black font-mono">
              <span className="text-rose-400 flex items-center gap-0.5">❤️ HP</span>
              <span className="text-slate-200">{heroCurrentHp.toLocaleString()} / {heroStats.maxHp.toLocaleString()}</span>
            </div>
            <div className="h-2 w-full overflow-hidden rounded-md bg-slate-900 border border-rose-950">
              <div
                className="h-full bg-gradient-to-r from-rose-600 via-rose-500 to-red-400 transition-all duration-150"

          <div className="space-y-0.5">
            <div className="flex items-center justify-between text-[8px] font-black font-mono">
              <span className="text-cyan-400 flex items-center gap-0.5">💧 MP</span>
              <span className="text-slate-200">{heroCurrentMp.toLocaleString()} / {heroStats.maxMp.toLocaleString()}</span>

            <div className="h-2 w-full overflow-hidden rounded-md bg-slate-900 border border-cyan-950">
              <div
                className="h-full bg-gradient-to-r from-cyan-600 to-blue-400 transition-all duration-150"
                style={{ width: `${Math.max(0, Math.min(100, (heroCurrentMp / heroStats.maxMp) * 100))}%` }}
              />
            </div>

          <div className="space-y-0.5">
            <div className="flex items-center justify-between text-[8px] font-black font-mono">
              <span className="text-amber-400 flex items-center gap-0.5">⭐ EXP</span>
              <span className="text-slate-200">{expPercent}%</span>
            </div>
            <div className="h-1.5 w-full overflow-hidden rounded-md bg-slate-900 border border-amber-950">
              <div
                className="h-full bg-gradient-to-r from-amber-500 to-yellow-300 transition-all duration-200"
                style={{ width: `${expPercent}%` }}
              />
            </div>
          </div>
        </div>

        {/* Stage Clear Flash Banner */}
        <AnimatePresence>
          {stageClearBanner && (
            <motion.div
              initial={{ opacity: 0, scale: 0.7, y: -20 }}
              animate={{ opacity: 1, scale: 1.05, y: 0 }}
              exit={{ opacity: 0, scale: 1.2 }}
              className="absolute inset-x-0 top-12 z-30 flex justify-center pointer-events-none"
            >
              <div className="rounded-2xl border-2 border-amber-400 bg-slate-950/90 px-6 py-2 shadow-[0_0_30px_rgba(245,158,11,0.8)] backdrop-blur-md">
                <span className="font-black tracking-widest text-sm text-transparent bg-clip-text bg-gradient-to-r from-amber-300 via-yellow-200 to-amber-400 uppercase">
                  {stageClearBanner}
                </span>
              </div>
            </motion.div>
          )}
        </AnimatePresence>

        {/* Level Up Banner */}
        <AnimatePresence>
          {levelUpBanner && (
            <motion.div
              initial={{ opacity: 0, y: 30, scale: 0.8 }}
              animate={{ opacity: 1, y: 0, scale: 1 }}
              exit={{ opacity: 0, y: -20 }}
              className="absolute inset-x-0 top-20 z-30 flex justify-center pointer-events-none"
            >
              <div className="rounded-2xl border-2 border-cyan-400 bg-cyan-950/90 px-6 py-2 shadow-[0_0_30px_rgba(6,182,212,0.9)] backdrop-blur-md text-center">
                <p className="font-black text-sm text-cyan-200 uppercase tracking-widest">
                  ⭐ AWANS JEDNOSTKI! LEVEL UP! ⭐
                </p>
                <p className="text-[10px] font-bold text-amber-300 font-mono">
                  Poziom Bohatera: Lv.{levelUpBanner} (+1 Punkt Umiejętności)
                </p>
              </div>
            </motion.div>
          )}
        </AnimatePresence>

        {/* Floating Damage Numbers */}
        <div className="absolute inset-0 pointer-events-none z-20">
          {floatingTexts.map((f) => (
            <motion.div
              key={f.id}
              initial={{ opacity: 1, y: `${f.y}%`, x: `${f.x}%`, scale: f.isCrit ? 1.4 : 1 }}
              animate={{ opacity: 0, y: `${f.y - 20}%`, scale: f.isCrit ? 1.6 : 1.1 }}
              transition={{ duration: 0.85 }}
              className="absolute font-mono font-black drop-shadow-[0_2px_4px_rgba(0,0,0,0.9)]"
              style={{ color: f.color, left: `${f.x}%`, top: `${f.y}%` }}
            >
              {f.text}
            </motion.div>
          ))}
        </div>

        {/* Arena Stage Entities */}
        <div className="absolute inset-x-0 bottom-20 top-4 flex items-center justify-between px-6">
          {/* Left: Hunter Hero & Shadow Allies */}
          <div className="relative flex flex-col items-center">
            {/* Active Shadow Allies Roster */}
            <div className="absolute -left-4 -top-8 flex items-center gap-2 opacity-95">
              {shadowArmy.map((shadow) => {
                if (!shadow.unlocked) return null;
                return (
                  <motion.div
                    key={shadow.id}
                    animate={{ y: [0, -5, 0] }}
                    transition={{ repeat: Infinity, duration: 2.2 + Math.random(), ease: "easeInOut" }}
                    className="flex flex-col items-center"
                  >
                    <div className="h-11 w-11 rounded-2xl border-2 border-purple-400/60 bg-purple-950/70 p-0.5 shadow-[0_0_12px_rgba(168,85,247,0.7)] flex items-center justify-center">
                      <img
                        src={SPRITES.shadows[shadow.id]}
                        alt={shadow.name}
                        className="h-full w-full object-contain filter drop-shadow-[0_0_6px_#a855f7]"
                      />
                    </div>
                    <span className="mt-0.5 rounded bg-slate-950/90 px-1 py-0.2 text-[7px] font-black text-purple-300 border border-purple-500/40">
                      Lv.{shadow.level}
                    </span>
                  </motion.div>
                );
              })}
            </div>

            {/* Hunter Hero Model with forward attack physical dash animation */}
            <motion.div
              animate={{
                scale: heroAttacking ? 1.15 : 1,
                x: heroAttacking ? 45 : 0,
                rotate: heroAttacking ? -6 : 0,
              }}
              transition={{ duration: 0.12 }}
              className={`relative flex flex-col items-center ${heroHitFlash ? "filter brightness-150 contrast-125 hue-rotate-90" : ""}`}
            >
              <div className="h-36 w-36 flex items-center justify-center filter drop-shadow-[0_0_18px_rgba(6,182,212,0.9)]">
                <img src={currentHunterSrc} alt="Hunter Hero" className="h-full w-full object-contain" />
              </div>
            </motion.div>
          </div>

          {/* Center Combo Banner */}
          <div className="flex flex-col items-center justify-center">
            {comboCount > 0 && (
              <span className="font-mono text-xs font-black text-cyan-300 drop-shadow-[0_0_8px_rgba(6,182,212,0.9)] animate-pulse">
                COMBO x{comboCount}
              </span>
            )}
            {awakeningActive && (
              <span className="mt-1 text-[9px] font-black uppercase tracking-widest text-purple-300 drop-shadow-[0_0_10px_rgba(168,85,247,1)]">
                PRZEBUDZENIE MONARCHY
              </span>
            )}
          </div>

          {/* Right: Current Enemy / Boss with direct hit spark overlay */}
          <div className="relative flex flex-col items-center">
            <motion.div
              animate={{
                scale: enemyHitFlash ? 0.9 : 1,
                x: enemyHitFlash ? 12 : 0,
              }}
              transition={{ duration: 0.1 }}
              className={`relative flex flex-col items-center ${enemyHitFlash ? "filter brightness-200 contrast-150" : ""}`}
            >
              <div className={`${enemy.isBoss ? "h-40 w-40" : "h-32 w-32"} relative flex items-center justify-center filter drop-shadow-[0_0_18px_rgba(244,63,94,0.8)]`}>
                <img src={currentEnemySrc} alt={enemy.name} className="h-full w-full object-contain" />

                {/* Direct Slash Hit Spark on Enemy */}
                {slashSparkActive && (
                  <motion.div
                    initial={{ opacity: 0, scale: 0.5, rotate: -20 }}
                    animate={{ opacity: 1, scale: 1.2, rotate: 20 }}
                    exit={{ opacity: 0 }}
                    transition={{ duration: 0.15 }}
                    className="absolute inset-0 flex items-center justify-center pointer-events-none z-10"
                  >
                    <img
                      src={slashSparkActive}
                      alt="Slash Spark"
                      className="h-28 w-28 object-contain filter drop-shadow-[0_0_15px_#38bdf8]"
                    />
                  </motion.div>
                )}
              </div>

              {/* Enemy HP Bar */}
              <div className="w-40 space-y-1 mt-1 text-center">
                <span className="block truncate text-[9px] font-black uppercase tracking-wider text-rose-300">
                  {enemy.name}
                </span>
                <div className="h-2.5 w-full overflow-hidden rounded-full bg-slate-900 border border-rose-900/60">
                  <div
                    className="h-full bg-gradient-to-r from-rose-700 via-rose-500 to-amber-400 transition-all duration-150"
                    style={{ width: `${Math.max(0, Math.min(100, (enemy.currentHp / enemy.maxHp) * 100))}%` }}
                  />
                </div>
                <span className="font-mono text-[8px] font-bold text-slate-400">
                  {Math.max(0, enemy.currentHp).toLocaleString()} / {enemy.maxHp.toLocaleString()}
                </span>

          {skills.map((skill) => {
            const cooldownLeft = Math.max(

                    <p className="text-[8px] text-slate-400">
                      Zbiera nagrody wyłącznie gdy jesteś offline / poza grą!
                    </p>
                  </div>
                </div>

                <div className="text-right">
                  <p className="text-[9px] text-slate-400">Zysk offline:</p>
                  <p className="font-mono text-xs font-black text-amber-300">
                    +{afkSummary?.goldPerMinute || 0} 🪙 / +{afkSummary?.expPerMinute || 0} EXP/min
                  </p>
                </div>
              </div>

              <div className="mt-3 flex items-center justify-between border-t border-amber-500/20 pt-2.5">
                <div className="space-y-0.5">
                  <p className="text-[11px] font-black text-amber-200">
                    Zgromadzone nagrody: +{afkSummary?.goldEarned.toLocaleString() || 0} Złota, +{afkSummary?.expEarned || 0} EXP, +{afkSummary?.manaCrystalsEarned || 0} Kryształów
                  </p>
                  <p className="text-[8px] text-slate-400">Odebrany zysk trafia do profilu Solo Leveler!</p>
                </div>

                <div className="flex items-center gap-2">
                  {/* Upgrade Chest Button */}
                  <button
                    type="button"
                    onClick={handleUpgradeAfkChest}
                    className="rounded-2xl border border-amber-400/70 bg-slate-900 px-3 py-1.5 text-[10px] font-black uppercase tracking-wider text-amber-300 hover:bg-slate-800"
                  >
                    Ulepsz ({computeAfkChestUpgradeCost(saveState.afkChestLevel || 1).gold.toLocaleString()}🪙 {computeAfkChestUpgradeCost(saveState.afkChestLevel || 1).crystals}💎)
                  </button>

                  {/* Claim Button */}
                  <button
                    type="button"
                    onClick={handleClaimAfk}
                    className="rounded-2xl border border-amber-400 bg-gradient-to-r from-amber-500 to-amber-600 px-4 py-1.5 text-xs font-black uppercase tracking-wider text-slate-950 shadow-[0_0_20px_rgba(245,158,11,0.5)] transition active:scale-95 hover:brightness-110"
                  >
                    Odbierz
                  </button>
                </div>
              </div>
            </div>
          </div>
        )}
      </div>
    </div>
  );
}
