  name: "Eryk",
  level: 45,
  rank: "S",
  jobClass: "Cienisty Monarcha",
  stats: {
    strength: 80,
    agility: 60,
    vitality: 50,
    intelligence: 70,
    perception: 40,

    themeId: "system-dark",
    graphicsQuality: "balanced",
    soundEnabled: true,
    hapticsEnabled: true,
    volume: 1,
    voiceGuidance: true,
    miniGamesUnlocked: false,
    notifications: {} as any,
  } as any,
};

test("computeHeroStatsFromPlayer scales with percentage boosts and hero level", () => {
  const save = createDefaultIdleRpgSave();
  save.heroLevel = 10;
  const stats = computeHeroStatsFromPlayer(mockPlayer, save);
  assert.ok(stats.attack > 200, "ATK should scale above 200");
  assert.ok(stats.maxHp > 1000, "Max HP should scale above 1000");
  assert.ok(stats.critRate > 10, "Crit Rate should scale");
  assert.ok(stats.percentBoosts.strAttackPercent > 0, "STR % boost must be active");
  assert.ok(stats.percentBoosts.vitHpPercent > 0, "VIT % boost must be active");
});

test("computeShadowArmy includes 5 shadows with diamond unlock costs", () => {
  const save = createDefaultIdleRpgSave();
  const shadows = computeShadowArmy(mockPlayer, save);
  assert.equal(shadows.length, 5);
  const igris = shadows.find((s) => s.id === "igris_knight");
  assert.equal(igris?.unlocked, true);
  const assassin = shadows.find((s) => s.id === "shadow_assassin");
  assert.equal(assassin?.diamondUnlockCost, 500);
});

test("calculateOfflineAfkRewards calculates elapsed offline rewards up to chest capacity", () => {
  const save = createDefaultIdleRpgSave();
  save.stage = 10;
  save.afkChestLevel = 2; // 6 hours
  save.lastActiveExitAt = Date.now() - 3600 * 1000; // 1 hour offline
  const stats = computeHeroStatsFromPlayer(mockPlayer, save);
  const afk = calculateOfflineAfkRewards(save, stats);
  assert.ok(afk.goldEarned > 500, "Should earn more than 500 gold in 1 hour");
  assert.ok(afk.expEarned > 300, "Should earn more than 300 EXP in 1 hour");
  assert.equal(afk.isFull, false);
});

test("computeItemSellPrice calculates rarity price plus upgrade refund", () => {
  const item = {
    id: "test_1",
    name: "Sztylet",
    slot: "weapon" as const,
    rarity: "rare" as const,
    level: 2,
    attack: 50,
    defense: 0,
    hp: 0,
    critRate: 5,
    icon: "",
  };
  const price = computeItemSellPrice(item);
  assert.equal(price, 180 + 2 * 80);
});

test("getAfkChestMaxHours and computeAfkChestUpgradeCost scale", () => {
  assert.equal(getAfkChestMaxHours(1), 4);
  assert.equal(getAfkChestMaxHours(2), 6);
  assert.equal(getAfkChestMaxHours(7), 24);

  const cost = computeAfkChestUpgradeCost(1);
  assert.equal(cost.gold, 1500);
  assert.equal(cost.crystals, 20);
});
